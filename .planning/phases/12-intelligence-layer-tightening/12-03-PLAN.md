<objective>
Restore signal on coverage: kill the 100% INCOMPLETE flag bomb and replace it with a run-wide tiered embed-filter (D-32).

The baseline run flagged 100% of 509 tasks INCOMPLETE because every section with a single missed_item bombed every task in that section. 618 missed_items were reported but most were already-extracted or cross-section duplicates. Coverage stopped being useful as a reviewer signal.

Fix (per D-32):
1. **Drop** the per-node `TaskFlag.INCOMPLETE` auto-append in orchestrator.py (lines 286-288).
2. **Add** a post-dedup orchestrator step that embeds each missed_item against the final task corpus (titles + first AC condition) using the dedup agent's MiniLM embedder. Tier each miss:
   - `≥ 0.85` → DROP (covered, just a duplicate of an extracted task)
   - `0.70 - 0.85` → keep, `tier="likely_overlap"`
   - `< 0.70` → keep, `tier="uncovered"`
3. **Overwrite** `coverage_reports.json` with the filtered+tiered version.

Purpose: Move INCOMPLETE flag rate below 5% of tickets (INT-01) and make `coverage_reports.json` a reviewer-grade signal — no cross-section duplicates, no covered items mislabeled as missing (INT-05).

Output: One agent file modified (new `filter_against_corpus` method + `tier` field on MissedItem), one orchestrator step rewritten, one new pytest file. The new behavior is the new default — no env flag.

    ```python
    def filter_against_corpus(
        self,
        section_reports: dict[str, dict],
        final_tasks: list,                # list[ManagedTask] — typed loosely to avoid circular import
        embedder=None,                    # SentenceTransformer or compatible; required at runtime
        drop_threshold: float = COVERAGE_TIER_DROP_THRESHOLD,
        overlap_threshold: float = COVERAGE_TIER_OVERLAP_THRESHOLD,
    ) -> dict[str, dict]:
        """
        Filter per-section coverage reports against the FINAL dedup'd task corpus.

        For each missed_item across all sections:
          1. Compute its cosine similarity against the max over (final_task.title + first AC condition)
             for every task in final_tasks.
          2. If max_sim >= drop_threshold → DROP (already covered by an extracted task).
          3. If overlap_threshold <= max_sim < drop_threshold → keep with tier="likely_overlap".
          4. If max_sim < overlap_threshold → keep with tier="uncovered".

        Cross-section duplicates collapse naturally because every missed_item is scored
        against the same run-wide corpus — duplicates land in the same tier.

        Returns a NEW dict shaped like the input (section_id → SectionCoverageReport.model_dump),
        with the `missed_items` list filtered + tiered in place.

        Args:
            embedder: A SentenceTransformer (or any object with .encode(texts, normalize_embeddings=True)
                returning a (N, D) numpy array). The orchestrator passes the DedupAgent's embedder.
        """
        if not section_reports or embedder is None:
            return section_reports if not section_reports else {}

        # Build the corpus texts: title + first AC condition (per D-32 spec).
        corpus_texts: list[str] = []
        for t in final_tasks:
            first_ac = ""
            acs = getattr(t, "acceptance_criteria", None) or []
            for ac in acs:
                cond = getattr(ac, "condition", None)
                if cond:
                    first_ac = cond
                    break
            corpus_texts.append(f"{t.title} {first_ac}".strip())

        if not corpus_texts:
            # Nothing in the corpus to compare against — everything is "uncovered".
            return self._mark_all_uncovered(section_reports)

        # Encode the corpus once.
        import numpy as _np
        corpus_emb = embedder.encode(corpus_texts, normalize_embeddings=True)

        filtered: dict[str, dict] = {}
        for section_id, report in section_reports.items():
            missed = report.get("missed_items") or []
            if not missed:
                # Section had nothing to filter; preserve the entry as-is for completeness.
                filtered[section_id] = report
                continue

            descriptions = [m.get("description", "") for m in missed]
            miss_emb = embedder.encode(descriptions, normalize_embeddings=True)

            # Cosine similarity for normalized vectors == dot product.
            sims = _np.dot(miss_emb, corpus_emb.T)  # shape (len(missed), len(corpus))
            max_sims = sims.max(axis=1)

            kept: list[dict] = []
            for m, max_sim in zip(missed, max_sims):
                sim = float(max_sim)
                if sim >= drop_threshold:
                    # DROP — already covered.
                    self.audit.log(
                        run_id=self.run_id,
                        agent="CoverageChecker",
                        node_id=section_id,
                        action="COVERAGE_TIER_DROP",
                        detail=f"section={section_id} sim={sim:.3f} desc={(m.get('description') or '')[:80]}",
                    )
                    continue
                if sim >= overlap_threshold:
                    m_out = dict(m)
                    m_out["tier"] = "likely_overlap"
                    m_out["max_corpus_similarity"] = round(sim, 3)
                    kept.append(m_out)
                    self.audit.log(
                        run_id=self.run_id,
                        agent="CoverageChecker",
                        node_id=section_id,
                        action="COVERAGE_TIER_OVERLAP",
                        detail=f"section={section_id} sim={sim:.3f} desc={(m_out.get('description') or '')[:80]}",
                    )
                else:
                    m_out = dict(m)
                    m_out["tier"] = "uncovered"
                    m_out["max_corpus_similarity"] = round(sim, 3)
                    kept.append(m_out)

            if kept:
                report_out = dict(report)
                report_out["missed_items"] = kept
                filtered[section_id] = report_out
            # If kept is empty, the entire section's misses were all drops — omit the section entirely.

        return filtered

    @staticmethod
    def _mark_all_uncovered(section_reports: dict[str, dict]) -> dict[str, dict]:
        out: dict[str, dict] = {}
        for section_id, report in section_reports.items():
            missed = report.get("missed_items") or []
            tagged = []
            for m in missed:
                m_out = dict(m)
                m_out["tier"] = "uncovered"
                m_out["max_corpus_similarity"] = 0.0
                tagged.append(m_out)
            if tagged:
                report_out = dict(report)
                report_out["missed_items"] = tagged
                out[section_id] = report_out
        return out
    ```

    import numpy as np

    class _FakeEmbedder:
        """
        Deterministic 'embedder' for tests: maps each input text to a fixed unit vector
        based on a keyword tag. Allows direct control of cosine similarities.
        """
        def __init__(self, mapping: dict[str, list[float]]):
            # mapping: text-or-substring → 4d unit vector
            self.mapping = mapping

        def encode(self, texts, normalize_embeddings=True):
            out = []
            for t in texts:
                vec = None
                for key, v in self.mapping.items():
                    if key in t:
                        vec = np.array(v, dtype=np.float32)
                        break
                if vec is None:
                    vec = np.array([1.0, 0.0, 0.0, 0.0], dtype=np.float32)
                # Already unit-normalized by construction in the test fixtures.
                out.append(vec)
            return np.stack(out)
