    import json, sqlite3, sys

    int01_rate = (incomplete / n_tasks * 100.0) if n_tasks else 0.0

    # INT-04: LOW_CONFIDENCE flag rate
    low_conf = sum(1 for t in tasks if "LOW_CONFIDENCE" in (t.get("flags") or []))
    int04_rate = (low_conf / n_tasks * 100.0) if n_tasks else 0.0

    # Audit-DB metrics
    db = sqlite3.connect("data/audit.db")
    cur = db.cursor()

    cur.execute("SELECT COUNT(*) FROM audit_log WHERE run_id=? AND action='DEDUP_MERGE'", (run_id,))
    dedup_merges = cur.fetchone()[0]

    cur.execute(
        "SELECT COUNT(*) FROM audit_log WHERE run_id=? AND action IN ('EXTRACTION_ERROR','TASK_PARSE_ERROR')",
        (run_id,),
    )
    extraction_failures = cur.fetchone()[0]

    # Node count — derive from the pipeline_output.json if it has it, else from audit EXTRACTED rows
    cur.execute("SELECT COUNT(*) FROM audit_log WHERE run_id=? AND action='EXTRACTED'", (run_id,))
    extracted_node_count = cur.fetchone()[0]

    # CRITIQUE_FLAGGED conf check (INT-04 traceability)

        (run_id,),
    )
    critique_rows = cur.fetchall()
    import re
    zero_conf_count = 0
    nonzero_conf_count = 0
    for (detail,) in critique_rows:
        m = re.search(r"conf=([\d.]+)", detail or "")
        if m:
            c = float(m.group(1))
            if c == 0.0:
                zero_conf_count += 1
            else:
                nonzero_conf_count += 1

    # INT-05: tier coverage on coverage_reports.json (persisted post-filter)
    total_misses = sum(len((r or {}).get("missed_items") or []) for r in cov.values())
    tiered_misses = sum(
        1 for r in cov.values() for m in ((r or {}).get("missed_items") or [])

        1 for r in cov.values() for m in ((r or {}).get("missed_items") or [])

        1 for r in cov.values() for m in ((r or {}).get("missed_items") or [])
        if m.get("tier") == "uncovered"
    )

    # DIAGNOSTIC ROW 2 (12-RESEARCH §9): Dedup batch-halve frequency
    cur.execute("SELECT COUNT(*) FROM audit_log WHERE run_id=? AND action='DEDUP_BATCH_HALVE'", (run_id,))

    partial_count = cur.fetchone()[0]

    repair_count = cur.fetchone()[0]

    print("─── Phase 12 metrics for run", run_id, "───")
    print(f"INT-01 INCOMPLETE rate:      {incomplete}/{n_tasks} = {int01_rate:.1f}%  (baseline 100.0%, target <5%)")
    print(f"INT-02 dedup merges:         {dedup_merges}                       (baseline 0,     target ≥20)")
    print(f"INT-03 extraction failures:  {extraction_failures}/{node_count} = {int03_rate:.1f}%  (baseline 29.0%, target <2%)")
    print(f"INT-04 LOW_CONFIDENCE rate:  {low_conf}/{n_tasks} = {int04_rate:.1f}%   (baseline 29.0%, target <10%)")
    print(f"        CRITIQUE_FLAGGED conf=0: {zero_conf_count}, conf>0: {nonzero_conf_count} (baseline 227 conf=0, target 0)")
    print(f"INT-05 coverage tier covg.:  {tiered_misses}/{total_misses} have tier field; {drops_in_file} drops persisted (target 0 drops)")
    print()
    print("─── Diagnostic distributions (12-RESEARCH §9 additive rows) ───")
    print(f"Coverage tier histogram:     drop={audit_drop_count} (audit) | likely_overlap={likely_overlap_count} | uncovered={uncovered_count}")
    print(f"                             (healthy: all three buckets nonzero; pathological: one bucket holds >90%)")
    print(f"Dedup batch-halve frequency: {halve_count} halves across {batch_count} total batches")
    print(f"                             (healthy: halve_count < 5; ≥5 signals Gemini JSON regression at default batch=30)")
    print()
    print("─── Partial-recovery wiring (validates 12-01 Task 1.3 BLOCKER fix) ───")
    print(f"EXTRACTION_PARTIAL rows:     {partial_count}")
    print(f"JSON_REPAIR_RECOVERED rows:  {repair_count}")

    # Validate every EXTRACTION_PARTIAL node was visible to gap_recovery
    # (closes 12-ASSUMPTIONS GA-1: no dead-letter audit rows).

        "SELECT node_id FROM audit_log WHERE run_id=? AND action='EXTRACTION_PARTIAL'", (run_id,)
